// Генератор 100 физически просчитанных уровней function generate100Levels() { const TILE = 32; const levels = []; // Допустимые отклонения по высоте для нашего прыжка (в тайлах) const maxStepUp = -3; // Максимум на 3 тайла вверх const maxStepDown = 4; // Максимум на 4 тайла вниз for (let lvl = 1; lvl <= 100; lvl++) { let platforms = []; let coins = []; let hazards = []; let currentX = 0; let currentY = 9; // Ряд по вертикали (0-12) let levelLengthInTiles = 30 + Math.floor(Math.random() * 20); // Длина от 30 до 50 блоков // Стартовая площадка platforms.push({ x: 0, y: 352, w: 160, h: 48 }); currentX = 160; // Поэтапная генерация рельефа while (currentX < levelLengthInTiles * TILE) { let widthInTiles = 3 + Math.floor(Math.random() * 5); // Ширина платформы от 3 до 7 тайлов let pWidth = widthInTiles * TILE; // Вычисляем новую высоту с учётом физики прыжка let stepY = Math.floor(Math.random() * (maxStepDown - maxStepUp + 1)) + maxStepUp; currentY = Math.max(3, Math.min(10, currentY + stepY)); // Ограничиваем экран сверху и снизу let pY = currentY * TILE; platforms.push({ x: currentX, y: pY, w: pWidth, h: 18 }); // С вероятностью 50% ставим монетку над платформой if (Math.random() > 0.4) { coins.push({ x: currentX + pWidth / 2, y: pY - 32, r: 8, collected: false }); } // С вероятностью 20% добавляем шипы на платформу (если она достаточно широкая) if (Math.random() < 0.2 && pWidth > 96) { hazards.push({ x: currentX + 32, y: pY - 16, w: 32, h: 16 }); } // Делаем разрыв между платформами (яма) или стыкуем вплотную let gap = (Math.random() < 0.3) ? (2 + Math.floor(Math.random() * 3)) * TILE : 0; currentX += pWidth + gap; } // Финиш в конце уровня let finishLine = { x: currentX - TILE, y: (currentY - 2) * TILE, w: 32, h: 52 }; // Сохраняем сгенерированный уровень levels.push({ id: lvl, platforms, coins, hazards, finishLine }); } console.log(`Сгенерировано уровней: ${levels.length}`); return levels; } // Запуск генерации const generatedPack = generate100Levels();
